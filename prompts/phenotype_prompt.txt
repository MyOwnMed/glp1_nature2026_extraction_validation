SYSTEM_MSG = "You are an expert medical AI analyzing clinical notes for phenotype information."

DOMAIN_INSTRUCTIONS = """\
Look for Human Phenotype Ontology (HPO) terms and provide the name or synonym \
that best matches the described phenotype.
Each term in the HPO describes a phenotypic abnormality, such as Atrial septal defect.
If possible (and if it adequately matches the source) extract the name as it \
might be referred to in HPO.

For each phenotype, assess:
- severity (mild, moderate, severe) — leave null if not specified.
- clinical_status — active, resolved, improving, worsening, stable, or unknown.
  * "edema improving on diuretics" → clinical_status: "improving"
  * "history of seizures, now resolved" → clinical_status: "resolved"
- certainty — confirmed, suspected, possible, or ruled_out.
  * "possible sleep apnea" → certainty: "suspected"
  * "denies edema" → negated: true, certainty: "confirmed"

Focus on observable clinical signs and phenotypic abnormalities — leave disease \
diagnoses to the disease status extractor.

For each finding, ALWAYS structure them as objects with the "name" field \
to ensure compatibility with our medical concept mapping system."""

# ── Standardisation ───────────────────────────────────────────────────

def _standardize(data: Dict[str, Any]) -> Dict[str, Any]:
    section = data.get(SECTION_KEY, {})

    # mentioned → bool
    mentioned = section.get("mentioned")
    if isinstance(mentioned, str):
        section["mentioned"] = mentioned.lower() in ("true", "yes")
    elif mentioned is None:
        section["mentioned"] = False

    # ensure findings list
    if ITEMS_KEY not in section:
        section[ITEMS_KEY] = []
    elif not isinstance(section[ITEMS_KEY], list):
        section[ITEMS_KEY] = [section[ITEMS_KEY]]

    for i, f in enumerate(section[ITEMS_KEY]):
        if isinstance(f, str):
            section[ITEMS_KEY][i] = {"name": f}
            f = section[ITEMS_KEY][i]

        # Map legacy field names
        if "phenotype" in f and "name" not in f:
            f["name"] = f.pop("phenotype")
        # Map old "status" (which was severity) → severity
        if "status" in f and f.get("severity") is None:
            old_status = str(f.pop("status", "")).lower()
            if old_status in ("mild", "moderate", "severe"):
                f["severity"] = old_status

        # Apply common standardisation
        standardize_common_fields(f)

        # Tag for HPO postprocessor
        f["phenotype"] = True

        # Fallback atomic_statement
        if "atomic_statement" not in f and not f.get("carry_forward"):
            name = f["name"]
            if f["negated"]:
                f["atomic_statement"] = f"The patient does not have {name} — it was explicitly denied or absent."
            elif f.get("severity"):
                f["atomic_statement"] = (
                    f"The patient presents with {name}, assessed as {f['severity']} severity"
                    f" (clinical status: {f['clinical_status']})."
                )
            elif f["clinical_status"] != "unknown":
                f["atomic_statement"] = f"The patient has {name} (clinical status: {f['clinical_status']})."
            else:
                f["atomic_statement"] = f"The patient has {name}."

    data[SECTION_KEY] = section
    return data
